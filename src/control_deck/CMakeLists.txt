add_library(control_deck SHARED Service.cpp Service.hpp Editor.hpp Sound.cpp Sound.hpp Hotkeys.cpp LoudnessSession.cpp LoudnessSession.hpp
    "${PROJECT_SOURCE_DIR}/src/ui/src/ControlDeckControls.cpp"
    "${PROJECT_SOURCE_DIR}/src/ui/src/ControlDeckEditor.cpp"
    "${PROJECT_SOURCE_DIR}/src/ui/src/ControlDeckLayout.cpp"
    "${PROJECT_SOURCE_DIR}/src/ui/src/ControlDeckOverlay.cpp")
wjn_configure_shared_target(control_deck WJN_CONTROL_DECK_BUILD)
target_include_directories(control_deck PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}"
    "${PROJECT_SOURCE_DIR}/src/common/include")
target_link_libraries(control_deck PRIVATE control_deck_state WinJACKNexus::UI meterbridge_ui_model PUBLIC Qt6::Widgets)
target_compile_options(control_deck PRIVATE /utf-8)
target_compile_definitions(control_deck PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)

set(control_deck_kinds mono stereo 5_1 7_1 midi sound peak peak_stereo peak_5_1 peak_7_1
    gain_mono gain_stereo gain_5_1 gain_7_1)
foreach(type oscilloscope spectrogram polar spectrum_bars spectrum_curve loudness)
    foreach(layout mono stereo 5_1 7_1)
        list(APPEND control_deck_kinds ${type}_${layout})
    endforeach()
endforeach()
foreach(kind IN LISTS control_deck_kinds)
    set(control 0)
    set(relative_folder "routing/${kind}")
    set(category "Control Deck / Routing")
    if(kind MATCHES "^(oscilloscope|spectrogram|polar|spectrum_bars|spectrum_curve|loudness)_(mono|stereo|5_1|7_1)$")
        set(analysis_type "${CMAKE_MATCH_1}")
        set(analysis_layout "${CMAKE_MATCH_2}")
        if(analysis_layout STREQUAL "mono")
            set(channels 1)
        elseif(analysis_layout STREQUAL "stereo")
            set(channels 2)
        elseif(analysis_layout STREQUAL "5_1")
            set(channels 6)
        else()
            set(channels 8)
        endif()
        if(analysis_type STREQUAL "oscilloscope")
            set(control 4)
            set(analysis_name "Oscilloscope")
        elseif(analysis_type STREQUAL "spectrogram")
            set(control 5)
            set(analysis_name "Spectrogram")
        elseif(analysis_type STREQUAL "polar")
            set(control 6)
            set(analysis_name "Polar Energy")
        elseif(analysis_type STREQUAL "spectrum_bars")
            set(control 7)
            set(analysis_name "Spectrum Bars")
        elseif(analysis_type STREQUAL "spectrum_curve")
            set(control 8)
            set(analysis_name "Spectrum Curve")
        else()
            set(control 9)
            set(analysis_name "Loudness")
        endif()
        set(display "Control Deck ${analysis_name} ${analysis_layout}")
        set(relative_folder "analysis/${analysis_type}/${analysis_layout}")
        set(category "Control Deck / Analysis / ${analysis_name}")
        if(analysis_type STREQUAL "loudness")
            set(relative_folder "meters/loudness/${analysis_layout}")
            set(category "Control Deck / Meters / Loudness")
        endif()
    elseif(kind STREQUAL "mono")
        set(channels 1)
        set(display "Signal Switch Mono")
    elseif(kind STREQUAL "stereo")
        set(channels 2)
        set(display "Signal Switch Stereo")
    elseif(kind STREQUAL "5_1")
        set(channels 6)
        set(display "Signal Switch 5.1")
    elseif(kind STREQUAL "7_1")
        set(channels 8)
        set(display "Signal Switch 7.1")
    elseif(kind STREQUAL "midi")
        set(channels 0)
        set(display "Signal Switch MIDI")
    elseif(kind STREQUAL "sound")
        set(channels 2)
        set(control 1)
        set(display "WJN Control Deck Sound")
    elseif(kind MATCHES "^peak")
        set(control 2)
        if(kind STREQUAL "peak")
            set(channels 1)
            set(display "WJN Control Deck Peak Meter")
        elseif(kind STREQUAL "peak_stereo")
            set(channels 2)
            set(display "WJN Control Deck Peak Meter Stereo")
        elseif(kind STREQUAL "peak_5_1")
            set(channels 6)
            set(display "WJN Control Deck Peak Meter 5.1")
        else()
            set(channels 8)
            set(display "WJN Control Deck Peak Meter 7.1")
        endif()
    else()
        set(control 3)
        if(kind STREQUAL "gain_mono")
            set(channels 1)
            set(display "WJN Control Deck Gain Mono")
        elseif(kind STREQUAL "gain_stereo")
            set(channels 2)
            set(display "WJN Control Deck Gain Stereo")
        elseif(kind STREQUAL "gain_5_1")
            set(channels 6)
            set(display "WJN Control Deck Gain 5.1")
        else()
            set(channels 8)
            set(display "WJN Control Deck Gain 7.1")
        endif()
    endif()
    if(control EQUAL 1)
        set(relative_folder "playback/sound")
        set(category "Control Deck / Playback")
    elseif(control EQUAL 2)
        if(kind STREQUAL "peak")
            set(relative_folder "meters/peak/mono")
        else()
            string(REGEX REPLACE "^peak_" "" layout_folder "${kind}")
            set(relative_folder "meters/peak/${layout_folder}")
        endif()
        set(category "Control Deck / Meters / Peak")
    elseif(control EQUAL 3)
        string(REGEX REPLACE "^gain_" "" layout_folder "${kind}")
        set(relative_folder "gain/${layout_folder}")
        set(category "Control Deck / Gain")
    endif()
    set(target wjn_signal_switch_${kind})
    string(REPLACE "_" "-" module_kind "${kind}")
    add_library(${target} SHARED Module.cpp)
    list(APPEND control_deck_all_targets ${target})
    if(control GREATER_EQUAL 4)
        list(APPEND control_deck_analysis_targets ${target})
    endif()
    wjn_configure_shared_target(${target} WJN_NODE_IMPLEMENTATION)
    target_compile_definitions(${target} PRIVATE WJN_SWITCH_CHANNELS=${channels}
        WJN_SWITCH_CONTROL=${control}
        "WJN_SWITCH_KIND=\"${module_kind}\""
        "WJN_SWITCH_NAME=\"${display}\"" "WJN_SWITCH_CATEGORY=\"${category}\""
        WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(${target} PRIVATE /utf-8)
    target_include_directories(${target} PRIVATE "${PROJECT_SOURCE_DIR}/src/node/include")
    target_link_libraries(${target} PRIVATE control_deck WinJACKNexus::Audio)
    foreach(config Debug Release RelWithDebInfo MinSizeRel)
        string(TOUPPER "${config}" upper)
        set_target_properties(${target} PROPERTIES
            RUNTIME_OUTPUT_DIRECTORY_${upper} "${WJN_OUTPUT_DIR}/nodes/control_deck/${relative_folder}")
    endforeach()
    add_custom_command(TARGET ${target} POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/${kind}/locate"
            "$<TARGET_FILE_DIR:${target}>/locate")
endforeach()
